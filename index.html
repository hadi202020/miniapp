import asyncio from aiogram import Bot, Dispatcher from aiogram.client.bot import DefaultBotProperties from aiogram.fsm.storage.memory import MemoryStorage from bot.config import BOT_TOKEN from bot.handlers.start import router as start_router from bot.handlers.plans import router as plans_router from bot.handlers.username import router as username_router from bot.handlers.payment import router as payment_router from bot.handlers import receipt, admin_payments from bot.handlers import wallet from bot.handlers.webapp import router as webapp_router async def main(): bot = Bot(token=BOT_TOKEN, default=DefaultBotProperties(parse_mode="HTML")) storage = MemoryStorage() # FSM در حافظه dp = Dispatcher(storage=storage) dp.include_router(start_router) dp.include_router(plans_router) dp.include_router(username_router) dp.include_router(payment_router) dp.include_router(admin_payments.router) dp.include_router(receipt.router) dp.include_router(wallet.router) dp.include_router(webapp_router) print("✅ Bot is polling...") await dp.start_polling(bot) if __name__ == "__main__": asyncio.run(main())
